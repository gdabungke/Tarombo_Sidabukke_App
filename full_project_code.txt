/**
 * FUNGSI KODE: main.ts
 * AUTHOR: Gindo Dabungke
 * Titik masuk klien: Sidebar Navigation, 4 Renderers (Tree, List, Gen, Table), Auth, & Settings.
 */
import { renderGenealogyTree } from "./treeRenderer.js";
import { renderGenealogyList } from "./listRenderer.js";
import { renderByGeneration } from "./generationRenderer.js";
import { renderGenealogyTable } from "./tableRenderer.js";

const API_BASE_URL = "http://localhost:5000";

console.log("Tarombo Raja Sidabukke (WarisKeluarga) Loaded successfully!");

document.addEventListener('DOMContentLoaded', () => {
  // Render awal silsilah berbentuk grafik D3.js
  renderGenealogyTree('#genealogy-svg');

  // 1. Sidebar Navigation Logic (Pindah Antar Menu)
  const navItems = document.querySelectorAll('.nav-item');
  const sections = document.querySelectorAll('main > section');
  const pageTitle = document.getElementById('page-title');

  const titlesMap: Record<string, string> = {
    'view-dashboard': 'Dashboard / Ringkasan Keluarga',
    'view-tree': 'Pohon Silsilah / Visualisasi Generasi',
    'view-members': 'Direktori Data Anggota Keluarga',
    'view-gallery': 'Galeri & Arsip Dokumentasi',
    'view-propose': 'Pangidoan Usulan & Koreksi Tarombo',
    'view-settings': 'Pengaturan & Info Pengembang'
  };

  navItems.forEach(item => {
    item.addEventListener('click', () => {
      const targetId = item.getAttribute('data-target');
      if (!targetId) return;

      navItems.forEach(nav => nav.classList.remove('active'));
      sections.forEach(sec => (sec as HTMLElement).style.display = 'none');

      item.classList.add('active');
      const targetSection = document.getElementById(targetId);
      if (targetSection) {
        targetSection.style.display = targetId === 'view-tree' ? 'flex' : 'block';
      }

      if (pageTitle && titlesMap[targetId]) {
        pageTitle.innerText = titlesMap[targetId];
      }

      if (targetId === 'view-tree') {
        renderGenealogyTree('#genealogy-svg');
      }
    });
  });

  // 2. Switch Desain Tampilan Silsilah (Grafik / Daftar / Generasi / Tabel)
  const btnTree = document.getElementById('btn-design-tree');
  const btnList = document.getElementById('btn-design-list');
  const btnGen = document.getElementById('btn-design-gen');
  const btnTable = document.getElementById('btn-design-table');

  const cGraphic = document.getElementById('container-tree-graphic');
  const cList = document.getElementById('container-tree-list');
  const cGen = document.getElementById('container-tree-gen');
  const cTable = document.getElementById('container-tree-table');

  const setActiveDesignButton = (activeBtn: HTMLElement | null) => {
    [btnTree, btnList, btnGen, btnTable].forEach(b => {
      if (b) { b.style.background = 'transparent'; b.style.color = 'var(--text-muted)'; }
    });
    if (activeBtn) { activeBtn.style.background = 'var(--accent-red)'; activeBtn.style.color = '#fff'; }
  };

  const hideAllContainers = () => {
    if (cGraphic) cGraphic.style.display = 'none';
    if (cList) cList.style.display = 'none';
    if (cGen) cGen.style.display = 'none';
    if (cTable) cTable.style.display = 'none';
  };

  if (btnTree && cGraphic) {
    btnTree.addEventListener('click', () => {
      setActiveDesignButton(btnTree);
      hideAllContainers();
      cGraphic.style.display = 'block';
      renderGenealogyTree('#genealogy-svg');
    });
  }

  if (btnList && cList) {
    btnList.addEventListener('click', () => {
      setActiveDesignButton(btnList);
      hideAllContainers();
      cList.style.display = 'block';
      renderGenealogyList('#container-tree-list');
    });
  }

  if (btnGen && cGen) {
    btnGen.addEventListener('click', () => {
      setActiveDesignButton(btnGen);
      hideAllContainers();
      cGen.style.display = 'block';
      renderByGeneration('#container-tree-gen');
    });
  }

  if (btnTable && cTable) {
    btnTable.addEventListener('click', () => {
      setActiveDesignButton(btnTable);
      hideAllContainers();
      cTable.style.display = 'block';
      renderGenealogyTable('#container-tree-table');
    });
  }

  // 3. Tombol Pengaturan di Header Atas
  const btnOpenSettingsTop = document.getElementById('btn-open-settings-top');
  if (btnOpenSettingsTop) {
    btnOpenSettingsTop.addEventListener('click', () => {
      const settingsNav = document.querySelector('[data-target="view-settings"]');
      if (settingsNav) (settingsNav as HTMLElement).click();
    });
  }

  // 4. Pengaturan Tema, Ukuran Huruf, & Bahasa
  const selectTheme = document.getElementById('select-theme') as HTMLSelectElement;
  const selectFontsize = document.getElementById('select-fontsize') as HTMLSelectElement;
  const selectLang = document.getElementById('select-lang') as HTMLSelectElement;

  if (selectTheme) {
    selectTheme.value = localStorage.getItem('tarombo_theme') || 'dark';
    selectTheme.addEventListener('change', () => {
      const theme = selectTheme.value;
      document.body.setAttribute('data-theme', theme);
      localStorage.setItem('tarombo_theme', theme);
    });
    document.body.setAttribute('data-theme', selectTheme.value);
  }

  if (selectFontsize) {
    selectFontsize.value = localStorage.getItem('tarombo_fontsize') || 'medium';
    selectFontsize.addEventListener('change', () => {
      const fs = selectFontsize.value;
      document.body.setAttribute('data-fontsize', fs);
      localStorage.setItem('tarombo_fontsize', fs);
    });
    document.body.setAttribute('data-fontsize', selectFontsize.value);
  }

  if (selectLang) {
    selectLang.value = localStorage.getItem('tarombo_lang') || 'id';
    selectLang.addEventListener('change', () => {
      const lang = selectLang.value;
      localStorage.setItem('tarombo_lang', lang);
      applyLanguage(lang);
    });
    applyLanguage(selectLang.value);
  }

  function applyLanguage(lang: string) {
    const txtPropTitle = document.getElementById('txt-prop-title');
    const txtPropDesc = document.getElementById('txt-prop-desc');
    const lblPropName = document.getElementById('lbl-prop-name');
    const lblPropReason = document.getElementById('lbl-prop-reason');
    const btnSubmitProp = document.getElementById('btn-submit-prop');
    const txtSetTitle = document.getElementById('txt-set-title');

    if (lang === 'batak') {
      if (txtPropTitle) txtPropTitle.innerText = "Pangidoan Usulan & Pangosean Tarombo";
      if (txtPropDesc) txtPropDesc.innerText = "Pomparan boi mamboan usulan manang koreksi data silsilah na naeng dipatigor pengurus.";
      if (lblPropName) lblPropName.innerText = "Goar na Diusulkan / Koreksi";
      if (lblPropReason) lblPropReason.innerText = "Alasan / Hubungan Tarombo";
      if (btnSubmitProp) btnSubmitProp.innerText = "Kirim Usulan";
      if (txtSetTitle) txtSetTitle.innerText = "⚙️ Pangaturan dohot Barita Pengembang";
    } else {
      if (txtPropTitle) txtPropTitle.innerText = "Pengajuan Usulan & Perubahan Silsilah";
      if (txtPropDesc) txtPropDesc.innerText = "Keluarga dapat mengajukan usulan penambahan data atau koreksi nama tarombo yang akan diverifikasi oleh pengurus.";
      if (lblPropName) lblPropName.innerText = "Nama yang Diusulkan / Koreksi";

import * as d3 from "d3";

export async function renderGenealogyTree(svgSelector: string) {
  try {
    const res = await fetch('/api/genealogy/tree');
    if (!res.ok) {
      throw new Error(`HTTP error! status: ${res.status}`);
    }

    const json = await res.json();
    const rootData = json.data;

    if (!rootData) {
      console.warn("Data silsilah kosong atau tidak ditemukan.");
      return;
    }

    const svg = d3.select(svgSelector);
    svg.selectAll("*").remove();

    const width = 1000;
    const height = 600;

    svg
      .attr("viewBox", `0 0 ${width} ${height}`)
      .attr("width", "100%")
      .attr("height", "100%");

    const g = svg.append("g").attr("transform", "translate(80, 50)");

    // Zoom behavior
    const zoom = d3.zoom<SVGSVGElement, unknown>().on("zoom", (event) => {
      g.attr("transform", event.transform);
    });
    svg.call(zoom as any);

    const treeLayout = d3.tree().nodeSize([70, 180]);
    const root = d3.hierarchy(rootData, (d: any) => d.children);

    treeLayout(root as any);

    // Links (garis penghubung)
    g.selectAll(".link")
      .data(root.links())
      .enter()
      .append("path")
      .attr("class", "link")
      .attr("fill", "none")
      .attr("stroke", "var(--border-color, #4b5563)")
      .attr("stroke-width", 1.5)
      .attr("d", d3.linkHorizontal<any, any>()
        .x((d: any) => d.y)
        .y((d: any) => d.x) as any
      );

    // Nodes
    const node = g.selectAll(".node")
      .data(root.descendants())
      .enter()
      .append("g")
      .attr("class", "node")
      .attr("transform", (d: any) => `translate(${d.y},${d.x})`)
      .style("cursor", "pointer")
      .on("click", (_event, d: any) => {
        const item = d.data;
        alert(`Detail Ompu / Pomparan:\n\nNama: ${item.fullName || '-'}\nGenerasi Ke: ${item.generationLevel || '-'}\nStatus: ${item.isAlive ? 'Mahidup' : 'Almarhum'}\nKeterangan: ${item.notes || '-'}`);
      });

    // Node Circle
    node.append("circle")
      .attr("r", 6)
      .attr("fill", (d: any) => d.data.isAlive ? "var(--accent-red, #ef4444)" : "#6b7280")
      .attr("stroke", "#fff")
      .attr("stroke-width", 1.5);

    // Node Label (Nama)
    node.append("text")
      .attr("dy", ".35em")
      .attr("x", (d: any) => d.children ? -12 : 12)
      .attr("text-anchor", (d: any) => d.children ? "end" : "start")
      .text((d: any) => d.data.fullName || "Tanpa Nama")
      .attr("fill", "var(--text-main, #f3f4f6)")
      .style("font-size", "12px")
      .style("font-family", "inherit");

  } catch (err) {
    console.error("Gagal merender pohon silsilah:", err);
  }
}

export function sanitizePersonData<T extends Partial<Person> & { biography?: Partial<Biography> | null }>(person: T, isLoggedIn: boolean): T {
  if (isLoggedIn) return person;
  if (person.isAlive) {
    return {
      ...person,
      fullName: 'Pomparan (Privat)',
      nickname: null,
      birthDateEst: null,
      deathDateEst: null,
      notes: '[Barita dipeakhon - Masuk ma tu akun keluarga asa diida]',
      biography: null,
    };
  }
  return person;
}

export function sanitizePersonsList<T extends any>(persons: T[], isLoggedIn: boolean): T[] {
  return persons.map((p) => sanitizePersonData(p, isLoggedIn));
}

import { Request, Response, NextFunction } from "express";
import jwt from "jsonwebtoken";
export type Role = "ADMIN" | "USER" | "SUPERADMIN" | string;

export interface UserPayload { userId: string; username: string; role: Role; }

declare global { namespace Express { interface Request { user?: UserPayload; } } }

const JWT_SECRET = process.env.JWT_SECRET || "tarombo_sidabukke_secure_random_key_2026";

export const authenticateToken = (req: Request, res: Response, next: NextFunction): void => {
  const token = req.headers['authorization']?.split(' ')[1];
  if (!token) { res.status(401).json({ message: 'Token ndang dapot.' }); return; }
  jwt.verify(token, JWT_SECRET, (err, decoded) => {
    if (err) { res.status(403).json({ message: 'Token ndang ture.' }); return; }
    req.user = decoded as UserPayload;
    next();
  });
};

export const optionalAuthenticateToken = (req: Request, _res: Response, next: NextFunction): void => {
  const authHeader = req.headers['authorization'];
  const token = authHeader?.split(' ')[1];

  // Abaikan jika tidak ada token, atau isinya string kosong / "undefined" / "null"
  if (!token || token === 'undefined' || token === 'null') {
    return next();
  }

  jwt.verify(token, JWT_SECRET, (err, decoded) => {
    if (!err && decoded) {
      req.user = decoded as UserPayload;
    }
    next();
  });
};

export const requireRole = (roles: Role[]) => (req: Request, res: Response, next: NextFunction): void => {
  if (!req.user || !roles.includes(req.user.role)) {
    res.status(403).json({ message: 'Akses ndang olo (ditolak).' });
    return;
  }
  next();
};

import { PrismaClient } from "@prisma/client";
import { sanitizePersonData } from "../../middlewares/privacy.js";

const prisma = new PrismaClient();

export async function buildGenealogyTree(rootPersonId?: string, isLoggedIn: boolean = false) {
  let rootId = rootPersonId;
  if (!rootId) {
    const root = await prisma.person.findFirst({ where: { generationLevel: 1 } });
    if (!root) return null;
    rootId = root.id;
  }
  return fetchNodeRecursive(rootId, isLoggedIn, new Set());
}

async function fetchNodeRecursive(id: string, isLoggedIn: boolean, visited: Set<string>): Promise<any> {
  if (visited.has(id)) return null;
  visited.add(id);

  const person = await prisma.person.findUnique({
    where: { id },
    include: {
      childRelations: { include: { child: true }, orderBy: { orderIndex: 'asc' } },
    },
  });

  if (!person) return null;
  const sanitized = sanitizePersonData(person, isLoggedIn);
  const children = [];

  for (const rel of person.childRelations) {
    const ch = await fetchNodeRecursive(rel.childId, isLoggedIn, new Set(visited));
    if (ch) children.push(ch);
  }

  return { ...sanitized, children };
}

import { Request, Response } from "express";
import bcrypt from "bcrypt";
import jwt from "jsonwebtoken";
import { PrismaClient, Role } from "@prisma/client";
import { registerSchema, loginSchema, changePasswordSchema } from "./auth.dto.js";

const prisma = new PrismaClient();
const JWT_SECRET = process.env.JWT_SECRET || "tarombo_sidabukke_secure_random_key_2026";

export async function register(req: Request, res: Response): Promise<void> {
  const parse = registerSchema.safeParse(req.body);
  if (!parse.success) { res.status(400).json({ errors: parse.error.format() }); return; }
  const { username, phone, email, password } = parse.data;
  try {
    const hashedPassword = await bcrypt.hash(password, 10);
    const user = await prisma.user.create({
      data: { username, phone, email: email || null, password: hashedPassword, role: Role.MEMBER },
      select: { id: true, username: true, phone: true, email: true, role: true, profilePic: true },
    });
    const token = jwt.sign({ userId: user.id, username: user.username, role: user.role }, JWT_SECRET, { expiresIn: '7d' });
    res.status(201).json({ message: 'Marhasil mendaftar.', token, user });
  } catch (e) {
    res.status(500).json({ message: 'Ndang tagamon mendaftar.' });
  }
}

export async function login(req: Request, res: Response): Promise<void> {
  const parse = loginSchema.safeParse(req.body);
  if (!parse.success) { res.status(400).json({ errors: parse.error.format() }); return; }
  const { identifier, password } = parse.data;
  try {
    const user = await prisma.user.findFirst({
      where: { OR: [{ phone: identifier }, { username: identifier }, { email: identifier }] },
    });
    if (!user || !(await bcrypt.compare(password, user.password))) {
      res.status(401).json({ message: 'Sandi manang username sala.' }); return;
    }
    const token = jwt.sign({ userId: user.id, username: user.username, role: user.role }, JWT_SECRET, { expiresIn: '7d' });
    res.json({ message: 'Login marhasil.', token, user: { id: user.id, username: user.username, role: user.role, profilePic: user.profilePic } });
  } catch (e) {
    res.status(500).json({ message: 'Server error.' });
  }
}

export async function getMe(req: Request, res: Response): Promise<void> {
  if (!req.user) { res.status(401).json({ message: 'Unauthorized' }); return; }
  const user = await prisma.user.findUnique({ where: { id: req.user.userId }, select: { id: true, username: true, phone: true, role: true, profilePic: true } });
  res.json({ user });
}

export async function changePassword(req: Request, res: Response): Promise<void> {
  if (!req.user) { res.status(401).json({ message: 'Unauthorized' }); return; }
  const parse = changePasswordSchema.safeParse(req.body);
  if (!parse.success) { res.status(400).json({ errors: parse.error.format() }); return; }
  const { oldPassword, newPassword } = parse.data;
  try {
    const user = await prisma.user.findUnique({ where: { id: req.user.userId } });
    if (!user || !(await bcrypt.compare(oldPassword, user.password))) {
      res.status(400).json({ message: 'Password na leleng sala.' }); return;
    }
    const hashedNew = await bcrypt.hash(newPassword, 10);
    await prisma.user.update({ where: { id: user.id }, data: { password: hashedNew } });
    res.json({ message: 'Password marhasil diubah.' });
  } catch (e) {
    res.status(500).json({ message: 'Gagal mubah password.' });
  }
}

export async function deleteAccount(req: Request, res: Response): Promise<void> {
  if (!req.user) { res.status(401).json({ message: 'Unauthorized' }); return; }
  try {
    await prisma.user.delete({ where: { id: req.user.userId } });
    res.json({ message: 'Akun nunga diaphus. Usulan ni keluarga tading aman.' });
  } catch (e) {
    res.status(500).json({ message: 'Gagal manghapus akun.' });
  }
}

export async function updateProfilePic(req: Request, res: Response): Promise<void> {
  if (!req.user) { res.status(401).json({ message: 'Unauthorized' }); return; }
  if (!req.file) { res.status(400).json({ message: 'Gambar foto profil ndang dapot.' }); return; }

    res.json({ message: 'Foto profil marhasil diubah.', user: updatedUser });
  } catch (e) {
    res.status(500).json({ message: 'Gagal mubah foto profil.' });
  }
}
